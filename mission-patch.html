<script>

"use strict";

(() => {

  const $ = id => document.getElementById(id);

  const canvas = $("patch");

  const ctx = canvas.getContext("2d");

  const palettes = {

    space: {bg:"#142c40",ink:"#f3f2eb",accent:"#8bc9a0"},

    earth: {bg:"#204b3b",ink:"#f3f2eb",accent:"#d8bd79"},

    mono:  {bg:"#f3f2eb",ink:"#242925",accent:"#242925"}

  };


  let edition = 0;

  let variation = {spread:1,lift:0,size:1,turns:[0,25,-25],iris:0};


  function random(a,b) { return a+Math.random()*(b-a); }


  function strokePath(draw,color,width,fill) {

    ctx.beginPath();

    draw();

    if(fill) { ctx.fillStyle=fill; ctx.fill(); }

    if(color) {

      ctx.strokeStyle=color;

      ctx.lineWidth=width;

      ctx.stroke();

    }

  }


  function circle(x,y,r,fill,stroke,width=8) {

    strokePath(()=>ctx.arc(x,y,r,0,Math.PI*2),stroke,width,fill);

  }


  function label(text,y,color,maxWidth) {

    if(!text.trim()) return;

    ctx.fillStyle=color;

    ctx.textAlign="center";

    ctx.textBaseline="middle";

    let size=48;

    do {

      ctx.font=`700 ${size}px Arial,Helvetica,sans-serif`;

      if(ctx.measureText(text).width<=maxWidth) break;

      size--;

    } while(size>12);

    ctx.fillText(text,500,y);

  }


  function puzzle(x,y,r,angle,color,width) {

    ctx.save();

    ctx.translate(x,y);

    ctx.rotate(angle*Math.PI/180);

    ctx.scale(r/82,r/82);

    // A simple silhouette with two tabs and one socket.

    strokePath(()=>{

      ctx.moveTo(-39,-36);

      ctx.lineTo(-13,-36);

      ctx.bezierCurveTo(-25,-65,25,-65,13,-36);

      ctx.lineTo(39,-36);

      ctx.lineTo(39,-12);

      ctx.bezierCurveTo(68,-24,68,24,39,12);

      ctx.lineTo(39,38);

      ctx.lineTo(13,38);

      ctx.bezierCurveTo(25,10,-25,10,-13,38);

      ctx.lineTo(-39,38);

      ctx.closePath();

    },color,width*0.72,color);

    ctx.restore();

  }


  function arrow(cx,cy,rx,ry,start,end,color,width) {

    const a=start*Math.PI/180,b=end*Math.PI/180;

    strokePath(()=>ctx.ellipse(cx,cy,rx,ry,0,a,b),color,width);

    const x=cx+rx*Math.cos(b),y=cy+ry*Math.sin(b);

    const tangent=Math.atan2(ry*Math.cos(b),-rx*Math.sin(b));

    ctx.save();

    ctx.translate(x,y);

    ctx.rotate(tangent);

    strokePath(()=>{

      ctx.moveTo(-22,-14);

      ctx.lineTo(0,0);

      ctx.lineTo(-22,14);

    },color,width);

    ctx.restore();

  }


  function render() {

    const p=palettes[$("palette").value];

    const weight=Number($("weight").value);

    const road=Number($("road").value)/100;

    const eye=Number($("eye").value)/100;

    $("road-value").textContent=$("road").value+"%";

    $("eye-value").textContent=$("eye").value+"%";

    $("weight-value").textContent=weight;


    ctx.setTransform(1,0,0,1,0,0);

    ctx.clearRect(0,0,1050,1050);

    ctx.setTransform(1.05,0,0,1.05,0,0);

    ctx.lineCap="round";

    ctx.lineJoin="round";

    ctx.fillStyle=p.bg;

    ctx.fillRect(0,0,1000,1000);


    // Square patch border and protected label areas.

    ctx.strokeStyle=p.ink;

    ctx.lineWidth=12;

    ctx.strokeRect(30,30,940,940);

    ctx.lineWidth=2;

    ctx.strokeRect(48,48,904,904);

    label($("top").value,112,p.ink,800);

    label($("bottom").value,918,p.ink,800);


    const layouts={

      balanced:{rx:282,ry:232,cy:486,r:82},

      wide:{rx:314,ry:220,cy:486,r:78},

      compact:{rx:254,ry:210,cy:474,r:76}

    };

    const l=layouts[$("layout").value];

    const rx=l.rx*variation.spread;

    const ry=l.ry;

    const cy=l.cy+variation.lift;

    const r=l.r*variation.size;


    // Three arrow segments establish one clockwise cycle.

    arrow(500,cy,rx,ry,-63,3,p.accent,weight);

    arrow(500,cy,rx,ry,57,123,p.accent,weight);

    arrow(500,cy,rx,ry,177,243,p.accent,weight);


    // Road overlaps the bottom orbit, as in the original sketch.

    const half=164*road;

    strokePath(()=>{

      ctx.moveTo(500-half,856);

      ctx.lineTo(490,530);

      ctx.quadraticCurveTo(500,518,510,530);

      ctx.lineTo(500+half,856);

      ctx.closePath();

    },p.ink,weight,p.bg);


    ctx.save();

    ctx.setLineDash([24,23]);

    strokePath(()=>{

      ctx.moveTo(500,555);

      ctx.lineTo(500,852);

    },p.accent,Math.max(5,weight*.72));

    ctx.restore();


    // Three floating circles, each containing a puzzle piece.

    [-90,30,150].forEach((degrees,i)=>{

      const angle=degrees*Math.PI/180;

      const x=500+rx*Math.cos(angle);

      const y=cy+ry*Math.sin(angle);

      circle(x,y,r,p.bg,p.ink,weight);

      puzzle(x,y,r,variation.turns[i],p.accent,weight);

    });


    // Central eye: restrained linework that survives reduction.

    ctx.save();

    ctx.translate(500,471);

    ctx.scale(eye,eye);

    strokePath(()=>{

      ctx.moveTo(-116,0);

      ctx.quadraticCurveTo(0,-94,116,0);

      ctx.quadraticCurveTo(0,88,-116,0);

      ctx.closePath();

    },p.ink,weight,p.bg);

    circle(variation.iris,0,33,p.accent,p.ink,weight*.7);

    circle(variation.iris,0,14,p.ink);

    ctx.restore();


    $("edition").textContent=edition

      ? "Variation "+String(edition).padStart(2,"0")

      : "Starting design";

  }


  function generate() {

    edition++;

    variation={

      spread:random(.96,1.04),

      lift:random(-7,7),

      size:random(.92,1.03),

      turns:[random(-45,45),random(-65,65),random(-65,65)],

      iris:random(-4,4)

    };

    render();

    $("status").textContent="New variation ready. Download it to keep it.";

  }


  function reset() {

    $("top").value="BARUCH";

    $("bottom").value="SSEP";

    $("palette").value="space";

    $("layout").value="balanced";

    $("road").value=100;

    $("eye").value=100;

    $("weight").value=8;

    edition=0;

    variation={spread:1,lift:0,size:1,turns:[0,25,-25],iris:0};

    render();

    $("status").textContent="Starting design restored.";

  }


  // Set PNG pHYs metadata to 300 dpi (11811 pixels per metre).

  // Canvas dimensions stay 1050 × 1050; artwork bytes are unchanged.

  function png300dpi(bytes) {

    function crc32(data) {

      let crc=0xffffffff;

      for(const byte of data) {

        crc^=byte;

        for(let j=0;j<8;j++)

          crc=(crc>>>1)^((crc&1)?0xedb88320:0);

      }

      return (crc^0xffffffff)>>>0;

    }

    const chunk=new Uint8Array(21);

    const view=new DataView(chunk.buffer);

    view.setUint32(0,9);

    chunk.set([112,72,89,115],4); // pHYs

    view.setUint32(8,11811);

    view.setUint32(12,11811);

    chunk[16]=1;

    view.setUint32(17,crc32(chunk.subarray(4,17)));


    const parts=[bytes.subarray(0,8)];

    let offset=8,inserted=false;

    const source=new DataView(

      bytes.buffer,bytes.byteOffset,bytes.byteLength

    );

    while(offset+12<=bytes.length) {

      const length=source.getUint32(offset);

      const end=offset+length+12;

      if(end>bytes.length) throw new Error("Invalid PNG data");

      const type=String.fromCharCode(...bytes.subarray(offset+4,offset+8));

      if(type!=="pHYs") parts.push(bytes.subarray(offset,end));

      if(type==="IHDR"&&!inserted) {

        parts.push(chunk);

        inserted=true;

      }

      offset=end;

    }

    return new Blob(parts,{type:"image/png"});

  }


  async function download() {

    const button=$("download");

    button.disabled=true;

    try {

      const raw=await new Promise((resolve,reject)=>{

        canvas.toBlob(blob=>blob?resolve(blob):

          reject(new Error("PNG export unavailable")),"image/png");

      });

      const blob=png300dpi(new Uint8Array(await raw.arrayBuffer()));

      const url=URL.createObjectURL(blob);

      const a=document.createElement("a");

      a.href=url;

      a.download="mission-patch-"+String(edition).padStart(2,"0")+".png";

      document.body.append(a);

      a.click();

      a.remove();

      setTimeout(()=>URL.revokeObjectURL(url),30000);

      $("status").textContent=

        "PNG download started · 1050 × 1050 px · 300 dpi.";

    } catch(error) {

      $("status").textContent="Download failed: "+error.message;

    } finally {

      button.disabled=false;

    }

  }


  ["top","bottom","palette","layout","road","eye","weight"]

    .forEach(id=>$(id).addEventListener("input",render));

  $("generate").addEventListener("click",generate);

  $("reset").addEventListener("click",reset);

  $("download").addEventListener("click",download);

  render();

})();

</script>
